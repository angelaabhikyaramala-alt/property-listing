<script>
    // In-browser mock database storing all records (including Sold and Rented)
    let propertiesDB = [
        { id: 1, title: "Downtown Luxury Villa", location: "Downtown", property_type: "Villa", price: 450000, description: "Spacious 4 BHK villa with garden.", status: "Available" },
        { id: 2, title: "Suburban Modern Apartment", location: "Suburbs", property_type: "Apartment", price: 180000, description: "Cozy 2 BHK apartment near park.", status: "Available" },
        { id: 3, title: "Downtown Studio", location: "Downtown", property_type: "Studio", price: 120000, description: "Fully furnished compact studio.", status: "Sold" },
        { id: 4, title: "Business District Commercial Space", location: "Business District", property_type: "Commercial", price: 300000, description: "Prime location office space.", status: "Rented" }
    ];

    document.addEventListener('DOMContentLoaded', () => {
        renderVisitorView(propertiesDB);
    });

    // Filter and search functionality (Location, Type, Price range)
    function filterProperties() {
        const loc = document.getElementById('filterLocation').value.toLowerCase();
        const type = document.getElementById('filterType').value;
        const minP = parseFloat(document.getElementById('filterMinPrice').value) || 0;
        const maxP = parseFloat(document.getElementById('filterMaxPrice').value) || Infinity;

        const filtered = propertiesDB.filter(p => {
            // Requirement: Exclude 'Sold' and 'Rented' from visitor view, but keep them in database
            const isAvailable = p.status === 'Available';
            const matchesLocation = p.location.toLowerCase().includes(loc);
            const matchesType = type === "" || p.property_type === type;
            const matchesPrice = p.price >= minP && p.price <= maxP;

            return isAvailable && matchesLocation && matchesType && matchesPrice;
        });

        renderVisitorView(filtered);
    }

    // Display property listings to visitors
    function renderVisitorView(data) {
        const grid = document.getElementById('propertyGrid');
        grid.innerHTML = '';

        if (data.length === 0) {
            grid.innerHTML = '<p>No available properties found matching your search criteria.</p>';
            return;
        }

        data.forEach(p => {
            const card = document.createElement('div');
            card.className = 'property-card';
            card.innerHTML = `
                <h3>${p.title}</h3>
                <p><strong>Location:</strong> ${p.location}</p>
                <p><strong>Type:</strong> ${p.property_type}</p>
                <p><strong>Price:</strong> $${p.price.toLocaleString()}</p>
                <p><span class="badge bg-${p.status}">${p.status}</span></p>
                <p>${p.description}</p>
                <button onclick="loadIntoForm(${p.id})">Edit</button>
                <button class="btn-danger" onclick="deleteProperty(${p.id})">Delete</button>
            `;
            grid.appendChild(card);
        });
    }

    // Add or Update Property
    function saveProperty(e) {
        e.preventDefault();
        const id = document.getElementById('propId').value;
        const title = document.getElementById('title').value;
        const location = document.getElementById('location').value;
        const property_type = document.getElementById('property_type').value;
        const price = parseFloat(document.getElementById('price').value);
        const status = document.getElementById('status').value;
        const description = document.getElementById('description').value;

        if (id) {
            // Update existing
            const index = propertiesDB.findIndex(p => p.id == id);
            if (index !== -1) {
                propertiesDB[index] = { id: Number(id), title, location, property_type, price, status, description };
            }
        } else {
            // Add new record
            const newId = propertiesDB.length > 0 ? Math.max(...propertiesDB.map(p => p.id)) + 1 : 1;
            propertiesDB.push({ id: newId, title, location, property_type, price, status, description });
        }

        document.getElementById('propertyForm').reset();
        document.getElementById('propId').value = '';
        document.getElementById('formTitle').innerText = 'Add New Property';
        filterProperties();
    }

    function loadIntoForm(id) {
        const p = propertiesDB.find(item => item.id === id);
        if (!p) return;
        document.getElementById('propId').value = p.id;
        document.getElementById('title').value = p.title;
        document.getElementById('location').value = p.location;
        document.getElementById('property_type').value = p.property_type;
        document.getElementById('price').value = p.price;
        document.getElementById('status').value = p.status;
        document.getElementById('description').value = p.description;
        document.getElementById('formTitle').innerText = 'Edit Property #' + p.id;
    }

    function deleteProperty(id) {
        if (confirm('Are you sure you want to delete this property?')) {
            propertiesDB = propertiesDB.filter(p => p.id !== id);
            filterProperties();
        }
    }
</script>
